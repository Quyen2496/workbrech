-- BƯỚC 1: KHỞI TẠO CƠ SỞ DỮ LIỆU
CREATE DATABASE IF NOT EXISTS healthsync_db;
USE healthsync_db;

-- Xóa bảng cũ nếu tồn tại để dựng lại cấu trúc chuẩn
DROP TABLE IF EXISTS Prescriptions;
DROP TABLE IF EXISTS Appointments;
DROP TABLE IF EXISTS Doctors;
DROP TABLE IF EXISTS Patients;

-- BƯỚC 2: THIẾT KẾ CÁC BẢNG NỀN TẢNG (PATIENTS & DOCTORS)
CREATE TABLE Patients (
    patient_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(15) NOT NULL
);

CREATE TABLE Doctors (
    doctor_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    specialty VARCHAR(50)
);

-- BƯỚC 3: THIẾT KẾ BẢNG APPOINTMENTS (ĐÃ TÁI CẤU TRÚC CHUẨN ĐÚNG ACTIVITY DIAGRAM)
CREATE TABLE Appointments (
    appointment_id INT AUTO_INCREMENT PRIMARY KEY,
    patient_id INT NOT NULL,
    doctor_id INT NOT NULL,
    appointment_date DATETIME NOT NULL,
    
    -- Xóa is_active, thay bằng ENUM quản lý 5 trạng thái
    status ENUM('PENDING', 'CONFIRMED', 'CHECKED_IN', 'COMPLETED', 'CANCELLED') DEFAULT 'PENDING',
    
    -- Sử dụng DECIMAL cho tính toán tài chính chính xác (không dùng FLOAT/DOUBLE)
    deposit_amount DECIMAL(10, 2) DEFAULT 0.00,
    penalty_fee DECIMAL(10, 2) DEFAULT 0.00,
    cancel_reason VARCHAR(255) DEFAULT NULL,
    
    FOREIGN KEY (patient_id) REFERENCES Patients(patient_id),
    FOREIGN KEY (doctor_id) REFERENCES Doctors(doctor_id)
);

-- BƯỚC 4: TẠO BẢNG PRESCRIPTIONS (LƯU ĐƠN THUỐC)
CREATE TABLE Prescriptions (
    prescription_id INT AUTO_INCREMENT PRIMARY KEY,
    appointment_id INT UNIQUE NOT NULL, -- UNIQUE đảm bảo mối quan hệ 1-1 với Lịch hẹn
    medication_details TEXT NOT NULL,
    issued_date DATETIME NOT NULL,
    FOREIGN KEY (appointment_id) REFERENCES Appointments(appointment_id)
);

-- BƯỚC 5: TẠO TRIGGER NGĂN CHẶN CHÈN ĐƠN THUỐC KHI LỊCH HẸN CHƯA COMPLETED (RÀNG BUỘC TOÀN VẸN)
DELIMITER //
CREATE TRIGGER trg_CheckAppointmentBeforePrescription
BEFORE INSERT ON Prescriptions
FOR EACH ROW
BEGIN
    DECLARE app_status VARCHAR(20);
    
    SELECT status INTO app_status
    FROM Appointments
    WHERE appointment_id = NEW.appointment_id;
    
    IF app_status != 'COMPLETED' THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Lỗi nghiệp vụ: Không thể kê đơn thuốc cho lịch hẹn chưa ở trạng thái COMPLETED!';
    END IF;
END;
//
DELIMITER ;

-- ============================================================================
-- BƯỚC 7: CÁC CÂU LỆNH TRUY VẤN ĐỐI SOÁT KIỂM TRA (SELECT QUERIES)
-- ============================================================================

-- Query 1: Lấy danh sách bệnh nhân đã hoàn tất khám bệnh kèm chi tiết đơn thuốc
SELECT 
    p.full_name AS Ten_Benh_Nhan,
    d.full_name AS Bacs_Si_Kham,
    a.appointment_date AS Ngay_Kham,
    a.status AS Trang_Thai,
    pr.medication_details AS Chi_Tiet_Don_Thuoc
FROM Appointments a
JOIN Patients p ON a.patient_id = p.patient_id
JOIN Doctors d ON a.doctor_id = d.doctor_id
JOIN Prescriptions pr ON a.appointment_id = pr.appointment_id
WHERE a.status = 'COMPLETED';

-- Query 2: Báo cáo tài chính thu phạt tiền cọc từ các lịch hủy
SELECT 
    a.appointment_id AS Ma_Lich_Hen,
    p.full_name AS Ten_Benh_Nhan,
    a.deposit_amount AS Tien_Coc,
    a.penalty_fee AS Phid_Phat,
    (a.deposit_amount - a.penalty_fee) AS Tien_Tra_Lai_Khach,
    a.cancel_reason AS Ly_Do_Huy
FROM Appointments a
JOIN Patients p ON a.patient_id = p.patient_id
WHERE a.status = 'CANCELLED';
